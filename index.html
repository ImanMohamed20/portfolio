const useStoreCart = () => {
  const [cart, setCart] = useState(() =>
    JSON.parse(localStorage.getItem('esw_cart') || '[]')
  );

  useEffect(() => {
    localStorage.setItem('esw_cart', JSON.stringify(cart));
  }, [cart]);

  const syncOrder = async (payload) => {
    const res = await fetch('https://server-esw.up.railway.app/api/orders', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    });
    return res.json();
  };
  return { cart, setCart, syncOrder };
};
